```
	// Called every frame
	virtual void Tick(float DeltaTime) override;

	// 상호작용

	FName TargetLevel;
```

```
#include "TeleportActor.h"

ATeleportActor::ATeleportActor()
{
	PrimaryActorTick.bCanEverTick = true;
}

void ATeleportActor::BeginPlay()
{
	Super::BeginPlay();
}

void ATeleportActor::Tick(float DeltaTime)
{
	Super::Tick(DeltaTime);
}

void ATeleportActor::Interact()
{
	UE_LOG(LogTemp, Warning, TEXT("TeleportActor Interact!"));

	if (!TargetLevel.IsNone())
	{
		UGameplayStatics::OpenLevel(this, TargetLevel);
	}
	else
	{
		UE_LOG(LogTemp, Warning, TEXT("TargetLevel is not set!"));
	}
}
```

```
LittleTreeCharacter.cpp



#include "Interactable.h"
...
void ALittleTreeCharacter::Interact()
{
	UE_LOG(LogTemplateCharacter, Log, TEXT("E KEY - INTERACT!"));

	// 카메라 위치
	const FVector Start = FollowCamera->GetComponentLocation();

	// 카메라가 바라보는 방향
	const FVector ForwardVector = FollowCamera->GetForwardVector();

	// 상호작용 거리
	const float InteractionDistance = 500.0f;

	// Line Trace 끝 위치
	const FVector End = Start + (ForwardVector * InteractionDistance);

	FHitResult HitResult;

	// 자기 자신은 무시
	FCollisionQueryParams CollisionParams;
	CollisionParams.AddIgnoredActor(this);

	// Line Trace
	const bool bHit = GetWorld()->LineTraceSingleByChannel(
		HitResult,
		Start,
		End,
		ECC_Visibility,
		CollisionParams
	);

	// 디버그용 선
	DrawDebugLine(
		GetWorld(),
		Start,
		End,
		bHit ? FColor::Green : FColor::Red,
		false,
		2.0f,
		0,
		2.0f
	);

	if (bHit)
	{
		AActor* HitActor = HitResult.GetActor();

		if (HitActor && HitActor->GetClass()->ImplementsInterface(UInteractable::StaticClass()))
		{
			IInteractable* Interactable = Cast<IInteractable>(HitActor);

			if (Interactable)
			{
				Interactable->Interact();
			}
		}
	}
	else
	{
		UE_LOG(LogTemplateCharacter, Log, TEXT("INTERACT: Nothing hit"));
	}
}
```
- Interacatable을 가지고 있어야 e로 상호작용할 수 있는 오브젝트를 구분할 수 있다.
- 같은 블루프린트를 이용해 디테일 창에서 도착지(Target Level)만 바꾸면 계속해서 무한 재활용할 수 있다!
